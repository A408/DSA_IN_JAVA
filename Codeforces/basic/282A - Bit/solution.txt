import java.util.*;

public class Main {
    public static void main(String[] args) {
      Scanner sc = new Scanner(System.in);
      int w = sc.nextInt();
      int x =0;
      for(int i =0;i<w;i++){
      String s = sc.next();
      if (s.charAt(1) == '+') {
        x++;
      }
      else {
        x--;
      }
      }
      System.out.println(x);
      }}
